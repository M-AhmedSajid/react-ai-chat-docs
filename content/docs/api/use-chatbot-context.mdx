# useChatbotContext

`useChatbotContext()` gives components access to the state and actions provided by `ChatbotProvider`.

It is the main API to use when building a custom chatbot UI.

## Import

```tsx id="x6m2pr"
import { useChatbotContext } from "react-ai-chat";
```

## Basic usage

Call the hook from a component rendered inside `ChatbotProvider`:

```tsx id="q9k4vs"
import { ChatbotProvider, useChatbotContext } from "react-ai-chat";

function CustomChatbot() {
  const { messages, input, setInput, handleSubmit, isLoading } =
    useChatbotContext();

  return (
    <div>
      <p>Messages: {messages.length}</p>
    </div>
  );
}

export default function App() {
  return (
    <ChatbotProvider>
      <CustomChatbot />
    </ChatbotProvider>
  );
}
```

## Context values

The hook exposes the chatbot's state and actions.

### `messages`

The current conversation messages.

```ts id="v7r3nk"
messages;
```

Use this to render the conversation.

```tsx id="j2p8wm"
{
  messages.map((message) => <div key={message.id}>{message.role}</div>);
}
```

Each message follows the message structure used by the AI SDK.

### `input`

The current input value.

```ts id="c5m9xq"
input: string;
```

```tsx id="a8v2kp"
<input value={input} />
```

### `setInput`

Updates the input value.

```ts id="p4z7md"
setInput: (value: string) => void
```

```tsx id="n6w3rt"
<input value={input} onChange={(event) => setInput(event.target.value)} />
```

### `handleSubmit`

Submits the current input and starts a chat request.

```ts id="k9q2xf"
handleSubmit: () => void
```

Example:

```tsx id="s4m8vc"
<button onClick={handleSubmit}>Send</button>
```

When building a form, prevent the browser's default form submission:

```tsx id="y7p3nb"
<form
  onSubmit={(event) => {
    event.preventDefault();
    handleSubmit();
  }}
>
  <button type="submit">Send</button>
</form>
```

### `isLoading`

Indicates whether a response is currently being generated.

```ts id="r8c5mq"
isLoading: boolean;
```

Use it to disable controls or show a loading state:

```tsx id="f3x7vz"
<button disabled={isLoading}>{isLoading ? "Thinking..." : "Send"}</button>
```

### `isOpen`

Indicates whether the chatbot is currently open.

```ts id="m2k6wp"
isOpen: boolean;
```

### `setIsOpen`

Updates the chatbot's open state.

```ts id="h5v9qc"
setIsOpen: (open: boolean) => void
```

```tsx id="t7n4xs"
<button onClick={() => setIsOpen(true)}>Open chat</button>
```

Close it with:

```tsx id="d8p3rm"
<button onClick={() => setIsOpen(false)}>Close</button>
```

## Building a custom input

The context API makes it possible to replace the generated input with your own component.

```tsx id="z6q2ky"
function CustomInput() {
  const { input, setInput, handleSubmit, isLoading } = useChatbotContext();

  return (
    <form
      onSubmit={(event) => {
        event.preventDefault();
        handleSubmit();
      }}
    >
      <input
        value={input}
        onChange={(event) => setInput(event.target.value)}
        placeholder="Ask a question..."
      />

      <button type="submit" disabled={isLoading || !input.trim()}>
        {isLoading ? "Thinking..." : "Send"}
      </button>
    </form>
  );
}
```

## Building a custom message list

You can also control how messages are rendered.

```tsx id="w4m8sp"
function CustomMessages() {
  const { messages } = useChatbotContext();

  return (
    <div>
      {messages.map((message) => (
        <div key={message.id}>
          <strong>{message.role}</strong>

          <div>
            {message.parts.map((part, index) =>
              part.type === "text" ? <p key={index}>{part.text}</p> : null,
            )}
          </div>
        </div>
      ))}
    </div>
  );
}
```

This gives you control over the presentation while the context continues to manage the conversation.

## Building a custom trigger

You can use the open state to create your own trigger:

```tsx id="e3q7vn"
function CustomTrigger() {
  const { isOpen, setIsOpen } = useChatbotContext();

  return (
    <button onClick={() => setIsOpen(!isOpen)}>
      {isOpen ? "Close chat" : "Open chat"}
    </button>
  );
}
```

## Full custom UI

The context can be combined to create an entire chatbot interface:

```tsx id="b8r4kc"
import { ChatbotProvider, useChatbotContext } from "react-ai-chat";

function Chat() {
  const {
    messages,
    input,
    setInput,
    handleSubmit,
    isLoading,
    isOpen,
    setIsOpen,
  } = useChatbotContext();

  if (!isOpen) {
    return <button onClick={() => setIsOpen(true)}>Open chat</button>;
  }

  return (
    <section>
      <header>
        <h2>AI Assistant</h2>
        <button onClick={() => setIsOpen(false)}>Close</button>
      </header>

      <div>
        {messages.map((message) => (
          <div key={message.id}>
            {message.role}:{" "}
            {message.parts.map((part, index) =>
              part.type === "text" ? (
                <span key={index}>{part.text}</span>
              ) : null,
            )}
          </div>
        ))}
      </div>

      <form
        onSubmit={(event) => {
          event.preventDefault();
          handleSubmit();
        }}
      >
        <input
          value={input}
          onChange={(event) => setInput(event.target.value)}
        />

        <button type="submit" disabled={isLoading || !input.trim()}>
          Send
        </button>
      </form>
    </section>
  );
}

export default function App() {
  return (
    <ChatbotProvider apiEndpoint="/api/chat">
      <Chat />
    </ChatbotProvider>
  );
}
```

## Provider requirement

`useChatbotContext()` must be used inside a `ChatbotProvider`.

```tsx id="u5j9pd"
<ChatbotProvider>
  <CustomChatbot />
</ChatbotProvider>
```

Calling the hook outside the provider results in an error.

## When to use this API

Use `useChatbotContext()` when you need custom control over the chatbot UI.

For a complete ready-made interface, use `<Chatbot />`.

For an editable starting point, use the generated chatbot:

```bash id="r2v6mx"
npx react-ai-chat init
```

The generated components are built around `ChatbotProvider` and `useChatbotContext()`.
