# Chatbot

`Chatbot` is the ready-made UI component included with `react-ai-chat`.

```tsx
import { Chatbot } from "react-ai-chat";
import "react-ai-chat/style.css";

export default function App() {
  return <Chatbot />;
}
```

## Import

```tsx
import { Chatbot } from "react-ai-chat";
```

Import the stylesheet once when using the ready-made UI:

```tsx
import "react-ai-chat/style.css";
```

## Props

### `apiEndpoint`

```ts
apiEndpoint?: string
```

The server endpoint used for chat requests.

**Default:** `"/api/chat"`

```tsx
<Chatbot apiEndpoint="/api/assistant" />
```

Your server route must expose a compatible `POST` handler at this endpoint.

See [API Route](/docs/getting-started/api-route).

### `title`

```ts
title?: string
```

The title displayed in the chatbot header.

### `subtitle`

```ts
subtitle?: string
```

The subtitle displayed below the chatbot title.

### `triggerText`

```ts
triggerText?: string
```

Text displayed on the chatbot trigger.

### `triggerIcon`

```ts
triggerIcon?: React.ReactNode
```

Custom icon rendered inside the chatbot trigger.

```tsx
<Chatbot triggerIcon={<ChatIcon />} />
```

### `sendIcon`

```ts
sendIcon?: React.ReactNode
```

Custom icon used for the send action.

### `closeIcon`

```ts
closeIcon?: React.ReactNode
```

Custom icon used to close the chatbot.

### `position`

```ts
position?: "bottom-right" | "bottom-left" | "top-right" | "top-left"
```

Controls where the chatbot is positioned.

**Default:** `"bottom-right"`

```tsx
<Chatbot position="bottom-left" />
```

### `starterPrompts`

```ts
starterPrompts?: string[]
```

Questions displayed when the conversation is empty.

```tsx
<Chatbot
  starterPrompts={["What can you help me with?", "How does this work?"]}
/>
```

### `emptyStateText`

```ts
emptyStateText?: string
```

Text displayed when the conversation has no messages.

### `placeholder`

```ts
placeholder?: string
```

Placeholder displayed inside the message input.

### `starterPromptsLabel`

```ts
starterPromptsLabel?: string
```

Label displayed above the starter prompts.

### `initialOpen`

```ts
initialOpen?: boolean
```

Controls whether the chatbot is open when it first renders.

**Default:** `false`

```tsx
<Chatbot initialOpen />
```

### `themeMode`

```ts
themeMode?: "system" | "light" | "dark"
```

Controls the chatbot's color mode.

**Default:** `"system"`

```tsx
<Chatbot themeMode="dark" />
```

### `theme`

```ts
theme?: ChatbotTheme
```

Customizes the chatbot's theme tokens.

```tsx
<Chatbot
  theme={{
    primary: "#7c3aed",
  }}
/>
```

See [Theming](/docs/guides/theming) for details.

### `classNames`

```ts
classNames?: ChatbotClassNames
```

Provides custom CSS classes for supported chatbot elements.

```tsx
<Chatbot
  classNames={{
    container: "my-chatbot",
    trigger: "my-trigger",
    panel: "my-panel",
  }}
/>
```

### `onError`

```ts
onError?: (error: Error) => void
```

Called when the chatbot encounters an error.

```tsx
<Chatbot
  onError={(error) => {
    console.error(error);
  }}
/>
```

## Complete example

```tsx
import { Chatbot } from "react-ai-chat";
import "react-ai-chat/style.css";

export default function App() {
  return (
    <Chatbot
      apiEndpoint="/api/chat"
      title="AI Assistant"
      subtitle="Ask me anything"
      triggerText="Chat"
      position="bottom-right"
      initialOpen={false}
      starterPrompts={[
        "What can you help me with?",
        "Tell me more about this project",
      ]}
      emptyStateText="How can I help you?"
      placeholder="Type your question..."
      starterPromptsLabel="Try asking"
      themeMode="system"
      onError={(error) => {
        console.error(error);
      }}
    />
  );
}
```
